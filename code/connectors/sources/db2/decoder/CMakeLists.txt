cmake_minimum_required(VERSION 3.20)
project(aleltdecode VERSION 0.1.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

option(ALELTDECODE_TESTS "Build unit tests" ON)
option(ALELTDECODE_FUZZ "Build libFuzzer target (clang / clang-cl)" OFF)

set(ALELTDECODE_SRC src/framing.c src/rowimage.c)

function(aleltdecode_warnings tgt)
  if(MSVC)
    target_compile_options(${tgt} PRIVATE /W4 /WX /sdl /guard:cf /permissive-)
    get_target_property(_type ${tgt} TYPE)
    if(NOT _type STREQUAL "STATIC_LIBRARY")
      target_link_options(${tgt} PRIVATE /GUARD:CF /DYNAMICBASE /NXCOMPAT)
    endif()
  else()
    target_compile_options(${tgt} PRIVATE -Wall -Wextra -Werror -Wpedantic -Wvla -Wconversion
                           -Wsign-conversion -Wshadow -O2 -fstack-protector-strong -fno-strict-aliasing)
    if(NOT ALELTDECODE_FUZZ)
      target_compile_definitions(${tgt} PRIVATE _FORTIFY_SOURCE=2)
    endif()
    get_target_property(_ltype ${tgt} TYPE)
    if(_ltype STREQUAL "SHARED_LIBRARY" AND NOT APPLE)
      target_link_options(${tgt} PRIVATE -Wl,-z,relro -Wl,-z,now -Wl,-z,noexecstack)
    endif()
  endif()
endfunction()

# Shared library loaded by the Python binding (decoder/python/aleltdecode_cffi.py)
add_library(aleltdecode SHARED ${ALELTDECODE_SRC})
target_include_directories(aleltdecode PUBLIC include PRIVATE src)
target_compile_definitions(aleltdecode PRIVATE ALELTDECODE_BUILD_DLL)
aleltdecode_warnings(aleltdecode)

add_library(aleltdecode_static STATIC ${ALELTDECODE_SRC})
target_include_directories(aleltdecode_static PUBLIC include PRIVATE src)
aleltdecode_warnings(aleltdecode_static)

if(ALELTDECODE_TESTS)
  enable_testing()
  add_executable(test_aleltdecode tests/test_aleltdecode.c)
  target_link_libraries(test_aleltdecode PRIVATE aleltdecode_static)
  aleltdecode_warnings(test_aleltdecode)
  add_test(NAME test_aleltdecode COMMAND test_aleltdecode)
endif()

if(ALELTDECODE_FUZZ)
  add_executable(fuzz_aleltdecode fuzz/fuzz_aleltdecode.c ${ALELTDECODE_SRC})
  target_include_directories(fuzz_aleltdecode PRIVATE include src)
  target_compile_options(fuzz_aleltdecode PRIVATE -fsanitize=fuzzer,address)
  target_link_options(fuzz_aleltdecode PRIVATE -fsanitize=fuzzer,address)
endif()
